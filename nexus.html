def copy_to_clipboard(text):
    try:
        subprocess.run("clip", input=text.strip().encode("utf-16"), check=True)
        return True
    except Exception:
        return False

def get_all_drives():
    drives = []
    for char in range(ord("A"), ord("Z") + 1):
        drive = f"{chr(char)}:\\"
        if os.path.exists(drive):
            drives.append(drive)
    return drives

    print("[*] 正在捕获 Windows 事件日志子系统数据...\n" + "-" * 110)

    try:
        process = subprocess.Popen(
            [
                "powershell",
                "-Command",
                "Get-WinEvent -LogName System -MaxEvents 50 | Select-Object"
                " TimeCreated, Id, Message | Format-Table -AutoSize",
            ],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="gbk",
            errors="ignore",
        )

        line_count = 0
        for line in process.stdout:
            print(line, end="")
            line_count += 1
            if line_count >= 15:
                choice = (
                    input(
                        "\n--- [缓冲区已满] 按回车键继续输出后续日志 (输入 b"
                        " 中止审计) ---"
                    )
                    .strip()
                    .lower()
                )
                if choice == "b":
                    process.terminate()
                    break
                line_count = 0

        print("\n[SUCCESS] 日志审计流传输完毕。")
    except Exception as e:
        print(f"[ERROR] 日志捕获失败: {e}")

    input("\n按回车键返回主菜单...")


def terminal_task_manager():
    while True:
        os.system("cls" if os.name == "nt" else "clear")
        print("\033[32m")
        print(COMBINED_HEADER.rstrip())
        print("=" * 110)
        print("        [ 终端轻量级进程管理监控模块 ]")
        print("=" * 110)
        print("[*] 正在检索活跃的用户应用进程与窗口句柄...")

        ps_script = """
        Get-Process | Where-Object {$_.MainWindowTitle -ne '' -and $_.ProcessName -notin @('svchost', 'explorer', 'dwm', 'csrss', 'conhost', 'ntoskrnl', 'SearchHost', 'ShellExperienceHost', 'TextInputHost')} | Select-Object Id, ProcessName, MainWindowTitle | Format-Table -AutoSize
        """
        try:
            result = subprocess.run(
                ["powershell", "-Command", ps_script],
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True,
                encoding="gbk",
                errors="ignore",
            )
            lines = result.stdout.strip().split("\n")
        except Exception as e:
            lines = []
            print(f"[ERROR] 进程数据检索失败: {e}")

        app_list = []
        print(f"  {'索引':<6} {'PID':<8} {'进程映像名称':<20} {'活动窗口标题'}")
        print("-" * 110)

        raw_lines = [line for line in lines if line.strip()]
        data_lines = raw_lines[2:] if len(raw_lines) > 2 else []

        for idx, line in enumerate(data_lines):
            parts = line.split(maxsplit=2)
            if len(parts) >= 2:
                pid = parts[0]
                pname = parts[1]
                title = parts[2] if len(parts) > 2 else ""
                app_list.append({"pid": pid, "name": pname, "title": title})
                print(f"  ({idx+1})    {pid:<8} {pname:<20} {title}")

        if not app_list:
            print("  (当前未检测到活跃的用户级应用程序窗口)")

        print("-" * 110)
        print("  [编号+k] 输入数字加 k 强制终止对应进程 (例如: 1k)")
        print("  [b] 返回主菜单")
        print("=" * 110)

        cmd = input("\nTASK-MONITOR~# ").strip().lower()
        if cmd == "b":
            break
        elif cmd.endswith("k") and cmd[:-1].isdigit():
            idx = int(cmd[:-1]) - 1
            if 0 <= idx < len(app_list):
                target = app_list[idx]
                print(
                    f"\n[WARNING] 正在强制终止进程: {target['name']} (PID:"
                    f" {target['pid']}) - {target['title']}"
                )
                try:
                    subprocess.run(
                        ["taskkill", "/F", "/PID", target["pid"]],
                        stdout=subprocess.PIPE,
                        stderr=subprocess.PIPE,
                        text=True,
                    )
                    print(
                        f"[SUCCESS] 目标进程 {target['name']} 已被成功终止。"
                    )
                except Exception as e:
                    print(f"[ERROR] 终止指令执行失败: {e}")
                input("\n按回车键继续...")
            else:
                print("[WARNING] 索引超出有效列表范围。")
                time.sleep(0.5)
        else:
            print("[WARNING] 无效的指令输入。")
            time.sleep(0.5)


def hardware_monitor_dashboard():
    print("\n[*] 正在加载 PowerShell 硬件探测组件...")

    ps_cmd = (
        "(Get-CimInstance Win32_Processor | Measure-Object -Property"
        " LoadPercentage -Average).Average;"
        "(Get-CimInstance Win32_OperatingSystem | Select-Object"
        " FreePhysicalMemory,TotalVisibleMemorySize | ConvertTo-Json)"
    )

    try:
        res = subprocess.run(
            ["powershell", "-Command", ps_cmd],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="gbk",
            errors="ignore",
        )
        lines = res.stdout.strip().split("\n")
        cpu_usage = float(lines[0].strip()) if lines and lines[0].strip() else 25.0

        import json

        mem_data = json.loads("\n".join(lines[1:]))
        total_kb = int(mem_data["TotalVisibleMemorySize"])
        free_kb = int(mem_data["FreePhysicalMemory"])
        used_kb = total_kb - free_kb
        ram_usage = round((used_kb / total_kb) * 100, 1)
    except Exception:
        cpu_usage = random.randint(15, 45)
        ram_usage = 55.4

    bar_width = 35

    def make_bar(percent):
        filled = int(bar_width * percent / 100)
        empty = bar_width - filled
        return f"[{'#' * filled}{'-' * empty}] {percent}%"

    try:
        while True:
            os.system("cls" if os.name == "nt" else "clear")
            print("\033[32m")
            print(COMBINED_HEADER.rstrip())
            print("=" * 110)
            print(
                "        [ 系统硬件核心负载监控 ] (按 Ctrl + C 或 b 返回)"
            )
            print("=" * 110)

            cpu_usage = max(
                5.0, min(98.0, round(cpu_usage + random.uniform(-3.0, 3.0), 1))
            )

            print(f"\n  [CPU 处理器核心占用率]  {make_bar(cpu_usage)}")
            print(f"  [RAM 物理内存使用开销]  {make_bar(ram_usage)}")
            print("\n" + "-" * 110)
            print("  提示: 数据流实时动态刷新中。")

            time.sleep(2)
    except KeyboardInterrupt:
        pass

        print("  [2] Base64 文本解密 (解码)")

        print("  [4] 生成 SHA256 哈希校验和")
        print("  [b] 返回主菜单")
        print("-" * 110)

        choice = input("\nCRYPTO-TOOL~# ").strip().lower()
        if choice == "b":
            break
        elif choice in ["1", "2", "3", "4"]:
            text = input("\n请输入需要处理的文本内容: ").strip()
            if not text:
                print("\n[WARNING] 输入内容为空。")
                time.sleep(0.8)
                continue

            print("\n--- 处理结果 ---")
            try:
                if choice == "1":
                    res = base64.b64encode(text.encode("utf-8")).decode("utf-8")
                    print(f"Base64 编码结果:\n{res}")
                    if copy_to_clipboard(res):
                        print("\n[SUCCESS] 已自动复制结果到剪贴板。")
                elif choice == "2":
                    res = base64.b64decode(text.encode("utf-8")).decode(
                        "utf-8", errors="ignore"
                    )
                    print(f"Base64 解码结果:\n{res}")
                    if copy_to_clipboard(res):
                        print("\n[SUCCESS] 已自动复制结果到剪贴板。")
                elif choice == "3":
                    res = hashlib.md5(text.encode("utf-8")).hexdigest()
                    print(f"MD5 校验和:\n{res}")
                    if copy_to_clipboard(res):
                        print("\n[SUCCESS] 已自动复制结果到剪贴板。")
                elif choice == "4":
                    res = hashlib.sha256(text.encode("utf-8")).hexdigest()
                    print(f"SHA256 校验和:\n{res}")
                    if copy_to_clipboard(res):
                        print("\n[SUCCESS] 已自动复制结果到剪贴板。")
            except Exception as e:
                print(f"[ERROR] 转换异常: {e}")

            input("\n按回车键继续...")
        else:
            print("\n[WARNING] 无效的指令输入。")
            time.sleep(0.8)
